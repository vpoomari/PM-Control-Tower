    46→const EDITABLE = ["DRAFT", "REJECTED"];
    47→const STATUS_TABS = ["ALL", "DRAFT", "SUBMITTED", "UNDER_REVIEW", "APPROVED", "REJECTED", "LOCKED"] as const;
    48→
    49→interface EditorRow {
    50→  key: string; date: string; projectId: string; taskId: string; activity: string;
    51→  startTime: string; endTime: string; breakMinutes: number; hours: number;
    52→  overtime: number; billable: boolean; comment: string;
    53→}
    54→
    55→let rowSeq = 0;
    56→function newRow(date: string): EditorRow {
    57→  rowSeq += 1;
    58→  return { key: `r${Date.now()}-${rowSeq}`, date, projectId: "", taskId: "", activity: "", startTime: "", endTime: "", breakMinutes: 0, hours: 8, overtime: 0, billable: true, comment: "" };
    59→}
    60→
    61→function calcHours(start: string, end: string, breakMinutes: number): number | null {
    62→  if (!start || !end) return null;
    63→  const [sh, sm] = start.split(":").map(Number);
    64→  const [eh, em] = end.split(":").map(Number);
    65→  if ([sh, sm, eh, em].some((n) => Number.isNaN(n))) return null;
    66→  const mins = eh * 60 + em - (sh * 60 + sm) - (breakMinutes || 0);
    67→  return mins > 0 ? round2(mins / 60) : null;
    68→}
    69→
    70→function dayIso(weekStart: Date, index: number): string {
    71→  return addDays(weekStart, index).toISOString().slice(0, 10);
    72→}

    74→export default function TimesheetsView() {
    75→  const me = useMe();
    76→  const canApprove = hasPerm(me, "timesheet.approve");
    77→  const [weekStart, setWeekStart] = useState<Date>(() => startOfWeek(new Date()));

    79→
    80→  const { options: projectOptions } = useProjectOptions();

    82→  // My resource record (determines whether the personal editor is shown)
    83→  const [myResource, setMyResource] = useState<ResLite | null>(null);
    84→  const [resourceChecked, setResourceChecked] = useState(false);
    85→  useEffect(() => {

    87→    let alive = true;

    92→        setMyResource(mine ? { id: mine.id, name: mine.name, email: mine.email, userId: mine.userId } : null);
    93→      })
    94→      .catch(() => { /* no access — editor stays hidden */ })
    95→      .finally(() => { if (alive) setResourceChecked(true); });
    96→    return () => { alive = false; };
    97→  }, [me]);
    98→
    99→  // ---- Listing (scope: all for approvers, mine otherwise) ----
   100→  const scope = canApprove ? "all" : "mine";
   101→  const list = useApi<TSListData>(`/api/timesheets?scope=${scope}`);
   102→  useRealtimeRefetch(list.refetch, ["timesheet:submitted", "timesheet:approved", "actuals:changed"]);
   103→
   104→  const [statusTab, setStatusTab] = useState<(typeof STATUS_TABS)[number]>("ALL");
   105→  const registerRows = useMemo(() => {
   106→    const rows = list.data?.timesheets ?? [];
   107→    return statusTab === "ALL" ? rows : rows.filter((t) => t.status === statusTab);
   108→  }, [list.data, statusTab]);
   109→
   110→  // ---- My timesheet for the selected week ----
   111→  const mineForWeek = useMemo(() => {
   112→    if (!myResource) return null;
   113→    return (list.data?.timesheets ?? []).find((t) => t.resourceId === myResource.id && t.weekStart.slice(0, 10) === weekKey) ?? null;
   114→  }, [list.data, myResource, weekKey]);
   115→
   116→  const mineDetail = useApi<TSDetail>(mineForWeek ? `/api/timesheets/${mineForWeek.id}` : null);
   117→  const editable = Boolean(mineForWeek && EDITABLE.includes(mineForWeek.status));
   118→
   119→  // ---- Editor state ----
   120→  const [rows, setRows] = useState<EditorRow[]>([]);
   121→  const [weekComments, setWeekComments] = useState("");
   122→  const [loadedTsId, setLoadedTsId] = useState<string | null>(null);
   123→  const [saving, setSaving] = useState(false);
   124→
   125→  useEffect(() => {
   126→    const ts = mineDetail.data?.timesheet;
   127→    if (!ts || !editable) return;
   128→    if (loadedTsId === ts.id) return;
   129→    setRows(ts.entries.map((e) => {
   130→      rowSeq += 1;
   131→      return {
   132→        key: `e${e.id}-${rowSeq}`, date: e.entryDate.slice(0, 10), projectId: e.projectId ?? "",
   133→        taskId: e.taskId ?? "", activity: e.activity ?? "", startTime: e.startTime ?? "", endTime: e.endTime ?? "",
   134→        breakMinutes: e.breakMinutes ?? 0, hours: e.hours ?? 0, overtime: e.overtimeHours ?? 0,
   135→        billable: e.billable, comment: e.comments ?? "",
   136→      };
   137→    }));
   138→    setWeekComments(ts.comments ?? "");
   139→    setLoadedTsId(ts.id);
   140→  }, [mineDetail.data, editable, loadedTsId]);
   141→
   142→  // Fresh editor when there is no existing timesheet for the week
   143→  useEffect(() => {
   144→    if (mineForWeek === null && myResource) {
   145→      setRows([newRow(dayIso(weekStart, 0))]);
   146→      setWeekComments("");
   147→      setLoadedTsId(null);
   148→    }
   149→  }, [mineForWeek, myResource, weekStart]);
   150→
   151→  const updateRow = (key: string, patch: Partial<EditorRow>) =>
   152→    setRows((prev) => prev.map((r) => (r.key === key ? { ...r, ...patch } : r)));
   153→
   154→  const onTimeField = (row: EditorRow, patch: Partial<EditorRow>) => {
   155→    const next = { ...row, ...patch };
   156→    const auto = calcHours(next.startTime, next.endTime, next.breakMinutes);
   157→    if (auto !== null) next.hours = auto;
   158→    updateRow(row.key, next);
   159→  };
   160→
   161→  const totals = useMemo(() => {
   162→    const valid = rows.filter((r) => r.hours > 0);
   163→    const total = round2(valid.reduce((s, r) => s + r.hours, 0));
   164→    const overtime = round2(valid.reduce((s, r) => s + (r.overtime || 0), 0));
   165→    const billable = round2(valid.filter((r) => r.billable).reduce((s, r) => s + r.hours, 0));
   166→    return { total, regular: round2(Math.max(0, total - overtime)), overtime, billable, nonBillable: round2(total - billable), count: valid.length };
   167→  }, [rows]);
   168→
   169→  const buildEntries = useCallback(() => {
   170→    return rows
   171→      .filter((r) => r.hours > 0 || r.projectId || r.activity)
   172→      .map((r) => ({
   173→        entryDate: r.date,
   174→        projectId: r.projectId || null,
   175→        taskId: r.taskId || null,
   176→        activity: r.activity || null,
   177→        startTime: r.startTime || null,
   178→        endTime: r.endTime || null,
   179→        breakMinutes: r.breakMinutes || 0,
   180→        hours: r.hours || 0,
   181→        overtimeHours: r.overtime || 0,
   182→        billable: r.billable,
   183→        comments: r.comment || null,
   184→      }));
   185→  }, [rows]);
   186→
   187→  const validate = (): string | null => {
   188→    const entries = buildEntries();
   189→    if (!entries.length) return "Add at least one entry with hours before saving.";
   190→    const withHours = entries.filter((e) => e.hours > 0);
   191→    if (!withHours.length) return "Entries need hours greater than zero.";
   192→    if (withHours.some((e) => !e.projectId)) return "Every entry with hours must reference a project.";
   193→    for (const e of withHours) {
   194→      if (e.overtimeHours && e.overtimeHours > (e.hours ?? 0)) return "Overtime cannot exceed the entry hours.";
   195→      if ((e.startTime && !e.endTime) || (!e.startTime && e.endTime)) return "Entries with times need both start and end (or clear both).";
   196→    }
   197→    return null;
   198→  };
   199→
   200→  const saveDraft = async (thenSubmit = false): Promise<string | null> => {
   201→    const err = validate();
   202→    if (err) { toast.error(err); return null; }
   203→    if (!myResource) return null;
   204→    setSaving(true);
   205→    try {
   206→      const body = { resourceId: myResource.id, weekStart: weekStart.toISOString(), comments: weekComments || null, entries: buildEntries() };
   207→      let tsId = loadedTsId;
   208→      if (mineForWeek && EDITABLE.includes(mineForWeek.status)) tsId = mineForWeek.id;
   209→      if (tsId) {
   210→        await api.patch(`/api/timesheets/${tsId}`, { comments: weekComments || null, entries: buildEntries() });
   211→      } else {
   212→        const d = await api.post<{ timesheet: { id: string } }>("/api/timesheets", body);
   213→        tsId = d.timesheet.id;
   214→      }
   215→      toast.success(thenSubmit ? "Saved — submitting…" : "Draft saved");
   216→      if (thenSubmit && tsId) {
   217→        await api.post(`/api/timesheets/${tsId}/submit`);
   218→        toast.success("Timesheet submitted for approval");
   219→      }
   220→      setLoadedTsId(tsId);
   221→      await list.refetch();
   222→      if (tsId) await mineDetail.refetch();
   223→      return tsId;
   224→    } catch (e) {
   225→      toast.error(e instanceof Error ? e.message : "Save failed");
   226→      return null;
   227→    } finally {
   228→      setSaving(false);
   229→    }
   230→  };
   231→
   232→  const deleteDraft = async () => {
   233→    if (!mineForWeek) return;
   234→    try {
   235→      await api.del(`/api/timesheets/${mineForWeek.id}`);
   236→      toast.success("Draft deleted");
   237→      setRows([newRow(dayIso(weekStart, 0))]);
   238→      setLoadedTsId(null);
   239→      await list.refetch();
   240→    } catch (e) {
   241→      toast.error(e instanceof Error ? e.message : "Delete failed");
   242→    }
   243→  };
   244→
   245→  // ---- Approvals ----
   246→  const [approvalView, setApprovalView] = useState<"PENDING" | "APPROVED">("PENDING");
   247→  const pending = useMemo(() => (list.data?.timesheets ?? []).filter((t) => ["SUBMITTED", "UNDER_REVIEW"].includes(t.status)), [list.data]);
   248→  const approved = useMemo(() => (list.data?.timesheets ?? []).filter((t) => t.status === "APPROVED"), [list.data]);
   249→  const [rejectTarget, setRejectTarget] = useState<TSListItem | null>(null);
   250→  const [rejectReason, setRejectReason] = useState("");
   251→  const [detailId, setDetailId] = useState<string | null>(null);
   252→  const detail = useApi<TSDetail>(detailId);
   253→  const detailTs = detail.data?.timesheet ?? null;
   254→
   255→  // Task-name map for read-only tables (entries carry ids only)
   256→  const [taskLabels, setTaskLabels] = useState<Record<string, string>>({});
   257→  const [projectLabels, setProjectLabels] = useState<Record<string, string>>({});
   258→  useEffect(() => {
   259→    const map: Record<string, string> = {};
   260→    projectOptions.forEach((p) => { map[p.id] = p.code; });
   261→    setProjectLabels(map);
   262→  }, [projectOptions]);
   263→
   264→  const loadTaskLabels = useCallback(async (entries: TSEntry[]) => {
   265→    const projectIds = [...new Set(entries.map((e) => e.projectId).filter((p): p is string => Boolean(p)))];
   266→    const taskIds = [...new Set(entries.map((e) => e.taskId).filter((t): t is string => Boolean(t)))];
   267→    if (!taskIds.length) return;
   268→    for (const pid of projectIds) {
   269→      try {
   270→        const d = await api.get<{ items: { id: string; code: string; name: string }[] }>(`/api/projects/${pid}/tasks`);
   271→        setTaskLabels((prev) => {
   272→          const next = { ...prev };
   273→          d.items.forEach((t) => { next[t.id] = t.code; });
   274→          return next;
   275→        });
   276→      } catch { /* label lookup best-effort */ }
   277→    }
   278→  }, []);
   279→
   280→  useEffect(() => {
   281→    const ts = detail.data?.timesheet;
   282→    if (ts) void loadTaskLabels(ts.entries);
   283→  }, [detail.data, loadTaskLabels]);
   284→
   285→  useEffect(() => {
   286→    const ts = mineDetail.data?.timesheet;
   287→    if (ts) void loadTaskLabels(ts.entries);
   288→  }, [mineDetail.data, loadTaskLabels]);
   289→
   290→  const act = async (fn: () => Promise<unknown>, okMsg: string) => {
   291→    try {
   292→      await fn();
   293→      toast.success(okMsg);
   294→      await list.refetch();
   295→      if (detailId) await detail.refetch();
   296→      if (mineForWeek) await mineDetail.refetch();
   297→    } catch (e) {
   298→      toast.error(e instanceof Error ? e.message : "Action failed");
   299→    }
   300→  };
   301→
   302→  const counts = list.data?.statusCounts ?? {};
   303→
   304→  if (!me || (list.loading && !list.data)) return <LoadingBlock label="Loading timesheets…" />;
   305→  if (list.error && !list.data) return <ErrorBlock message={list.error} onRetry={list.refetch} />;
   306→
   307→  const showMine = Boolean(myResource) && resourceChecked;
   308→  const ts = mineDetail.data?.timesheet ?? null;
   309→  const locked = Boolean(mineForWeek && ["APPROVED", "LOCKED", "SUBMITTED", "UNDER_REVIEW"].includes(mineForWeek.status));
   310→
   311→  return (
   312→    <div className="space-y-5">
   313→      <PageHeader
   314→        title="Timesheets"
   315→        subtitle="Capture effort, submit for approval and govern the actuals that feed cost, EVM and health."
   316→        breadcrumb={["Home", "Execute", "Timesheets"]}
   317→        actions={<WeekNav weekStart={weekStart} onChange={(d) => { setWeekStart(d); setLoadedTsId(null); }} />}
   318→      />
   319→
   320→      {/* MY TIMESHEET */}
   321→      {showMine && (
   322→        <SectionCard
   323→          title="My timesheet"
   324→          description={`Filing as ${myResource?.name} · week of ${weekKey}`}
   325→          actions={
   326→            mineForWeek ? (
   327→              <div className="flex items-center gap-2">
   328→                <StatusChip status={mineForWeek.status} />
   329→                <Badge variant="outline" className="border-slate-200 text-slate-500">{num(mineForWeek.totalHours)}h</Badge>
   330→              </div>
   331→            ) : <Badge variant="outline" className="border-amber-200 bg-amber-50 text-amber-700">No timesheet yet — create one</Badge>
   332→          }
   333→        >
   334→          {mineDetail.loading && !mineDetail.data ? <LoadingBlock label="Loading your week…" /> : (
   335→            <>
   336→              {mineForWeek?.status === "REJECTED" && mineForWeek.rejectionReason && (
   337→                <div className="mb-3 rounded-md border border-red-200 bg-red-50 px-3 py-2 text-sm text-red-700">
   338→                  <span className="font-semibold">Rejected by {mineForWeek.approverName || "approver"}:</span> {mineForWeek.rejectionReason} — update your entries and resubmit.
   339→                </div>
   340→              )}
   341→              {locked && ts ? (
   342→                <ReadOnlyWeek ts={ts} projectLabels={projectLabels} taskLabels={taskLabels} />

   346→                    {DAY_HEADERS.map((label, idx) => {
   347→                      const iso = dayIso(weekStart, idx);
   348→                      const dayRows = rows.filter((r) => r.date === iso);
   349→                      const dayTotal = round2(dayRows.reduce((s, r) => s + (r.hours || 0), 0));

   354→                              <CalendarDays className="h-3.5 w-3.5 text-slate-400" />
   355→                              <span className="text-xs font-semibold text-slate-700">{label} <span className="text-slate-400 font-normal">{iso.slice(5)}</span></span>

   368→                              {dayRows.map((r) => (

   372→                                    <TaskSelect projectId={r.projectId} value={r.taskId} onChange={(v) => updateRow(r.key, { taskId: v })} />

   380→                                  <div className="grid grid-cols-3 md:grid-cols-6 xl:grid-cols-7 gap-2 items-center">
   381→                                    <div>
   382→                                      <label className="text-[10px] uppercase text-slate-400">Start</label>

   392→                                    </div>
   393→                                    <div>
   394→                                      <label className="text-[10px] uppercase text-slate-400">Hours {calcHours(r.startTime, r.endTime, r.breakMinutes) !== null && <span className="text-blue-500 normal-case">(auto)</span>}</label>

   398→                                      <label className="text-[10px] uppercase text-slate-400">Overtime</label>

Actuals cascade into project cost and EVM.`}>

   616→// ---- Read-only week renderer (submitted/approved/locked + detail sheet) ----
   617→function ReadOnlyWeek({ ts, projectLabels, taskLabels, actions }: {
   618→  ts: TSListItem & { entries: TSEntry[] };
   619→  projectLabels: Record<string, string>;
   620→  taskLabels: Record<string, string>;
   621→  actions?: ReactNode;
   622→}) {

   624→    <div>

   638→        <div className="rounded-lg border border-slate-200 overflow-auto max-h-80">
   639→          <table className="w-full text-sm">
   640→            <thead className="sticky top-0 z-[1]">

   666→    </div>
   667→  );
   668→}
